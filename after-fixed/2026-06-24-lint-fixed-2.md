func StartConfigWatcher(logger *zap.Logger, fileReloaders map[string]ReloadFunc) {
	if len(fileReloaders) == 0 {
		return
	}

	watcher, err := fsnotify.NewWatcher()
	if err != nil {
		logger.Error("failed to create config file watcher", zap.Error(err))
		return
	}

	for path := range fileReloaders {
		if path == "" {
			continue
		}
		if err := watcher.Add(path); err != nil {
			logger.Warn("failed to watch config file", zap.String("file", path), zap.Error(err))
		}
	}

	go func() {
		defer func() {
			if cerr := watcher.Close(); cerr != nil {
				fmt.Fprintf(os.Stderr, "close watcher: %v\n", cerr)
			}
		}()
		for {
			select {
			case event, ok := <-watcher.Events:
				if !ok {
					return
				}
				if event.Has(fsnotify.Write) || event.Has(fsnotify.Create) {
					fn, found := fileReloaders[event.Name]
					if !found {
						continue
					}
					logger.Info("config file changed, reloading", zap.String("file", event.Name))
					fn()
					if event.Has(fsnotify.Create) {
						// Re-watch after atomic symlink swap (Kubernetes ConfigMap)
						_ = watcher.Add(event.Name)
					}
				}
			case err, ok := <-watcher.Errors:
				if !ok {
					return
				}
				logger.Error("config watcher error", zap.Error(err))
			}
		}
	}()
}
